/**
 * Xaraash shared storage on Google Sheets + Drive (free).
 * Posts go to the "posts" sheet of the spreadsheet this script is attached to (it lives in your My Drive).
 * Photos go to the Drive folder "Xaraash photos" (shared as "anyone with the link can view").
 * You can edit or delete any row directly in the Sheet. Set the "status" cell to hidden to hide a post.
 *
 * Setup: see README, section "Keydin Google Drive".
 */
var SHEET_NAME = 'posts';
var FOLDER_NAME = 'Xaraash photos';
var HEADERS = ['id', 'createdAt', 'cat', 'city', 'priceUsd', 'title', 'user', 'photos', 'status', 'phone', 'editKey', 'acct'];
var COL = { id: 0, createdAt: 1, cat: 2, city: 3, priceUsd: 4, title: 5, user: 6, photos: 7, status: 8, phone: 9, editKey: 10, acct: 11 };
var CATS = ['cars', 'devices', 'real', 'furniture', 'services', 'fashion', 'animals', 'jobs', 'games', 'coaching', 'food', 'gardens', 'tourism', 'trips', 'events', 'arts', 'coding', 'lost', 'projects'];
var MAX_PHOTOS = 5;
var MAX_PHOTO_CHARS = 700000;   // ~500 KB per photo after base64
var MAX_WRITES_PER_MINUTE = 30; // whole site, protects your Drive quota
var LIST_LIMIT = 200;

// Run this once from the editor (Run > setup) to approve access to Sheets and Drive.
function setup() { sheet_(); folder_(); }

function sheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.setFrozenRows(1); }
  var head = sh.getRange(1, 1, 1, HEADERS.length).getValues()[0];
  if (head.join('|') !== HEADERS.join('|')) sh.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);
  return sh;
}
function folder_() {
  var it = DriveApp.getFoldersByName(FOLDER_NAME);
  return it.hasNext() ? it.next() : DriveApp.createFolder(FOLDER_NAME);
}
function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

function cleanPhone_(v) {
  var s = String(v || '').replace(/[^0-9+()\s-]/g, '').trim().slice(0, 20);
  return s.replace(/\D/g, '').length >= 6 ? s : '';
}

// Validates and trims an incoming post (add or update). Returns null when it should be rejected.
function clean_(b, withPhotos) {
  if (!b || typeof b !== 'object') return null;
  var title = String(b.title || '').replace(/[\u0000-\u001f]/g, ' ').trim().slice(0, 80);
  if (!title) return null;
  var cat = CATS.indexOf(b.cat) >= 0 ? b.cat : 'services';
  var city = String(b.city || '').replace(/[\u0000-\u001f<>]/g, '').trim().slice(0, 40);
  var price = b.priceUsd === null || b.priceUsd === '' || b.priceUsd === undefined ? '' : Number(b.priceUsd);
  if (price !== '' && (!isFinite(price) || price < 0 || price > 1e9)) return null;
  var name = String(b.name || '').replace(/[\u0000-\u001f<>]/g, '').trim().slice(0, 40);
  var phone = cleanPhone_(b.phone);
  var photos = [];
  if (withPhotos) {
    photos = Array.isArray(b.photos) ? b.photos.slice(0, MAX_PHOTOS) : [];
    for (var i = 0; i < photos.length; i++) {
      if (typeof photos[i] !== 'string' || photos[i].indexOf('data:image/jpeg;base64,') !== 0 || photos[i].length > MAX_PHOTO_CHARS) return null;
    }
  }
  var acct = b.acct === 'company' ? 'company' : 'person';
  return { acct: acct, title: title, cat: cat, city: city, price: price, name: name, phone: phone, photos: photos };
}

function findRow_(sh, id) {
  var last = sh.getLastRow();
  if (last < 2 || !id) return -1;
  var ids = sh.getRange(2, 1, last - 1, 1).getValues();
  for (var i = ids.length - 1; i >= 0; i--) if (String(ids[i][0]) === String(id)) return i + 2;
  return -1;
}

function doGet(e) {
  try {
    var sh = sheet_();
    var last = sh.getLastRow();
    if (last < 2) return json_({ ok: true, posts: [] });
    var from = Math.max(2, last - LIST_LIMIT + 1);
    var rows = sh.getRange(from, 1, last - from + 1, HEADERS.length).getValues();
    var posts = [];
    for (var i = rows.length - 1; i >= 0; i--) {
      var r = rows[i];
      if (r[COL.status] === 'hidden' || r[COL.status] === 'deleted') continue;
      var ids = [];
      try { ids = JSON.parse(r[COL.photos] || '[]'); } catch (x) { ids = []; }
      posts.push({ id: String(r[COL.id]), ts: new Date(r[COL.createdAt]).getTime(), cat: r[COL.cat], city: r[COL.city],
        priceUsd: r[COL.priceUsd] === '' ? null : Number(r[COL.priceUsd]), title: r[COL.title], user: r[COL.user], phone: r[COL.phone] || '', acct: r[COL.acct] === 'company' ? 'company' : 'person', photos: ids });
    }
    return json_({ ok: true, posts: posts });  // editKey is never sent
  } catch (err) {
    return json_({ ok: false, error: 'list_failed' });
  }
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    var body = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    var action = body.action;
    if (['add', 'update', 'delete', 'report', 'rate', 'sendCode', 'verifyCode'].indexOf(action) < 0) return json_({ ok: false, error: 'bad_action' });

    var cache = CacheService.getScriptCache();
    var key = 'rl' + Math.floor(Date.now() / 60000);
    var n = Number(cache.get(key) || 0);
    if (n >= MAX_WRITES_PER_MINUTE) return json_({ ok: false, error: 'rate_limited' });
    cache.put(key, String(n + 1), 120);

    if (action === 'sendCode') return sendCode_(body.phone);
    if (action === 'verifyCode') return verifyCode_(body.phone, body.code);

    if (action === 'report' || action === 'rate') {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var txt = function (v, m) { return String(v == null ? '' : v).replace(/[\u0000-\u001f]/g, ' ').trim().slice(0, m); };
      if (action === 'report') {
        var reasons = ['r1', 'r2', 'r3', 'r4', 'r5'];
        var reason = reasons.indexOf(body.reason) >= 0 ? body.reason : 'r5';
        var rs = ss.getSheetByName('reports') || ss.insertSheet('reports');
        if (rs.getLastRow() === 0) rs.appendRow(['createdAt', 'postId', 'reason', 'note']);
        rs.appendRow([new Date().toISOString(), "'" + txt(body.postId, 40), reason, "'" + txt(body.note, 300)]);
        return json_({ ok: true });
      }
      var stars = Math.round(Number(body.stars));
      if (!(stars >= 1 && stars <= 5)) return json_({ ok: false, error: 'invalid' });
      var ra = ss.getSheetByName('ratings') || ss.insertSheet('ratings');
      if (ra.getLastRow() === 0) ra.appendRow(['createdAt', 'stars', 'comment']);
      ra.appendRow([new Date().toISOString(), stars, "'" + txt(body.comment, 300)]);
      return json_({ ok: true });
    }

    var sh = sheet_();

    if (action === 'add') {
      var p = clean_(body, true);
      if (!p || !p.name || !p.city || !p.phone) return json_({ ok: false, error: 'invalid' });
      var pd = digits_(p.phone);
      if (!pd || cache.get('tok' + String(body.token || '')) !== pd) return json_({ ok: false, error: 'unverified' });
      var id = Utilities.getUuid().slice(0, 8), editKey = Utilities.getUuid(), ids = [];
      if (p.photos.length) {
        var f = folder_();
        for (var i = 0; i < p.photos.length; i++) {
          var bytes = Utilities.base64Decode(p.photos[i].split(',')[1]);
          var file = f.createFile(Utilities.newBlob(bytes, 'image/jpeg', id + '-' + i + '.jpg'));
          file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
          ids.push(file.getId());
        }
      }
      sh.appendRow([id, new Date(), p.cat, p.city, p.price, p.title, p.name || ('Member ' + id.slice(0, 4)), JSON.stringify(ids), 'visible', p.phone, editKey, p.acct]);
      // A verification token can create one post only. This reduces replay/abuse.
      cache.remove('tok' + String(body.token || ''));
      return json_({ ok: true, id: id, editKey: editKey });
    }

    // update / delete need the secret key that was returned when the post was created
    var row = findRow_(sh, body.id);
    if (row < 0) return json_({ ok: false, error: 'not_found' });
    var stored = String(sh.getRange(row, COL.editKey + 1).getValue() || '');
    if (!stored || stored !== String(body.key || '')) return json_({ ok: false, error: 'forbidden' });

    if (action === 'delete') {
      sh.getRange(row, COL.status + 1).setValue('deleted');
      return json_({ ok: true });
    }
    var u = clean_(body, false);
    if (!u) return json_({ ok: false, error: 'invalid' });
    var name = u.name || String(sh.getRange(row, COL.user + 1).getValue() || '');
    if (!u.city) return json_({ ok: false, error: 'invalid' });
    sh.getRange(row, COL.cat + 1, 1, 4).setValues([[u.cat, u.city, u.price, u.title]]);
    sh.getRange(row, COL.user + 1).setValue(name);
    sh.getRange(row, COL.acct + 1).setValue(u.acct);
    // the verified phone number of a post cannot be changed here
    return json_({ ok: true });
  } catch (err) {
    return json_({ ok: false, error: 'server_error' });
  } finally {
    try { lock.releaseLock(); } catch (x) {}
  }
}


// ---------- WhatsApp verification (Meta WhatsApp Cloud API) ----------
// Script properties to set (Project settings > Script properties):
//   WA_TOKEN     permanent access token
//   WA_PHONE_ID  the "Phone number ID" of your WhatsApp sender
//   WA_TEMPLATE  approved authentication template name (default xaraash_code)
//   WA_LANG           exact approved template language code (default en_US)
//   WA_API_VERSION    Graph API version (default v26.0)
//   WA_TEMPLATE_MODE  authentication, custom_body, or custom_body_url
// Keep these values in Script Properties, never in index.html.
function digits_(v) {
  var d = String(v || '').replace(/\D/g, '');
  return d.length >= 8 && d.length <= 15 ? d : '';
}
function hash_(code, phone) {
  var props = PropertiesService.getScriptProperties();
  var salt = props.getProperty('SALT');
  if (!salt) { salt = Utilities.getUuid(); props.setProperty('SALT', salt); }
  var bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, salt + ':' + phone + ':' + code);
  return bytes.map(function (b) { return ('0' + (b & 0xff).toString(16)).slice(-2); }).join('');
}
function sendCode_(phone) {
  var d = digits_(phone);
  if (!d) return json_({ ok: false, error: 'bad_phone' });

  var props = PropertiesService.getScriptProperties();
  var token = props.getProperty('WA_TOKEN');
  var pid = props.getProperty('WA_PHONE_ID');
  if (!token || !pid) return json_({ ok: false, error: 'wa_not_configured' });

  var cache = CacheService.getScriptCache();
  var now = Date.now();
  var hourKey = 'sc' + d + Math.floor(now / 3600000);
  var hourCount = Number(cache.get(hourKey) || 0);

  // Maximum 3 sends per phone in a rolling hour bucket, plus a 60-second resend wait.
  if (hourCount >= 3) return json_({ ok: false, error: 'too_many' });
  if (cache.get('cool' + d)) return json_({ ok: false, error: 'wait_before_resend' });

  // Count the attempt before calling Meta, so API failures cannot be abused for unlimited retries.
  cache.put(hourKey, String(hourCount + 1), 3700);
  cache.put('cool' + d, '1', 60);

  // Only one active code per phone at a time.
  cache.remove('code' + d);
  cache.remove('att' + d);

  var code = ('00000' + (parseInt(Utilities.getUuid().replace(/-/g, '').slice(0, 8), 16) % 1000000)).slice(-6);
  var apiVersion = props.getProperty('WA_API_VERSION') || 'v26.0';
  var templateName = props.getProperty('WA_TEMPLATE') || 'xaraash_code';
  var lang = props.getProperty('WA_LANG') || 'en_US';
  var templateMode = String(props.getProperty('WA_TEMPLATE_MODE') || 'authentication').toLowerCase();

  var components;
  if (templateMode === 'authentication') {
    // Use an approved Meta AUTHENTICATION template with its OTP/copy-code URL button.
    components = [
      { type: 'body', parameters: [{ type: 'text', text: code }] },
      { type: 'button', sub_type: 'url', index: '0', parameters: [{ type: 'text', text: code }] }
    ];
  } else if (templateMode === 'custom_body') {
    // Use a separately approved custom template whose BODY contains a {{1}} variable.
    components = [
      { type: 'body', parameters: [{ type: 'text', text: code }] }
    ];
  } else if (templateMode === 'custom_body_url') {
    // Use a separately approved custom template with a BODY variable and a URL button variable.
    components = [
      { type: 'body', parameters: [{ type: 'text', text: code }] },
      { type: 'button', sub_type: 'url', index: '0', parameters: [{ type: 'text', text: code }] }
    ];
  } else {
    return json_({ ok: false, error: 'bad_template_mode' });
  }

  var response;
  try {
    response = UrlFetchApp.fetch('https://graph.facebook.com/' + apiVersion + '/' + encodeURIComponent(pid) + '/messages', {
      method: 'post',
      contentType: 'application/json',
      muteHttpExceptions: true,
      headers: { Authorization: 'Bearer ' + token },
      payload: JSON.stringify({
        messaging_product: 'whatsapp',
        to: d,
        type: 'template',
        template: {
          name: templateName,
          language: { code: lang },
          components: components
        }
      })
    });
  } catch (err) {
    return json_({ ok: false, error: 'send_failed' });
  }

  var status = response.getResponseCode();
  if (status < 200 || status >= 300) {
    // Do not return Meta's raw response to the public client because it may expose internal details.
    return json_({ ok: false, error: 'send_failed' });
  }

  cache.put('code' + d, hash_(code, d), 600);
  return json_({ ok: true, expiresIn: 600, resendAfter: 60 });
}
function verifyCode_(phone, code) {
  var d = digits_(phone);
  var c = String(code || '').replace(/\D/g, '');
  if (!d || c.length !== 6) return json_({ ok: false, error: 'invalid' });

  var cache = CacheService.getScriptCache();
  var attemptKey = 'att' + d;
  var attempts = Number(cache.get(attemptKey) || 0);
  if (attempts >= 5) return json_({ ok: false, error: 'too_many' });

  var stored = cache.get('code' + d);
  if (!stored) return json_({ ok: false, error: 'expired' });

  if (stored !== hash_(c, d)) {
    cache.put(attemptKey, String(attempts + 1), 600);
    return json_({ ok: false, error: attempts + 1 >= 5 ? 'too_many' : 'wrong_code' });
  }

  cache.remove('code' + d);
  cache.remove(attemptKey);

  var verificationToken = Utilities.getUuid() + Utilities.getUuid().replace(/-/g, '');
  cache.put('tok' + verificationToken, d, 3600);
  return json_({ ok: true, token: verificationToken, expiresIn: 3600 });
}
